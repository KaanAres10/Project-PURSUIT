using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;

public class VolumetricFogFeature : ScriptableRendererFeature
{
    public enum InjectionPoint
    {
        BeforeRenderingTransparents = RenderPassEvent.BeforeRenderingTransparents,
        BeforeRenderingPostProcessing = RenderPassEvent.BeforeRenderingPostProcessing,
        AfterRenderingPostProcessing  = RenderPassEvent.AfterRenderingPostProcessing
    }

    [Header("Execution")]
    public InjectionPoint injectionPoint = InjectionPoint.BeforeRenderingPostProcessing;

    [Tooltip("Material using shader \"Unlit/VolumetricFog\"")]
    public Material fogMaterial;

    [Tooltip("Shader pass index (usually 0).")]
    public int passIndex = 0;

    [Header("Buffers")]
    [Tooltip("Copy active color so Blit.hlsl (_BlitTexture/_BlitTexture_TexelSize) is valid.")]
    public bool fetchColorBuffer = true;

    [Tooltip("Bind camera depth-stencil while drawing (recommended ON).")]
    public bool bindDepthStencilAttachment = true;

    private FullscreenFogPass _pass;

    public override void Create()
    {
        _pass = new FullscreenFogPass(name);
    }

    public override void AddRenderPasses(ScriptableRenderer renderer, ref RenderingData renderingData)
    {
        if (UniversalRenderer.IsOffscreenDepthTexture(in renderingData.cameraData) ||
            renderingData.cameraData.cameraType == CameraType.Preview ||
            renderingData.cameraData.cameraType == CameraType.Reflection)
            return;

        if (fogMaterial == null)
        {
            Debug.LogWarning($"[{nameof(VolumetricFogFeature)}] Assign a fog material (shader \"Unlit/VolumetricFog\").");
            return;
        }

        if (passIndex < 0 || passIndex >= fogMaterial.passCount)
        {
            Debug.LogWarning($"[{nameof(VolumetricFogFeature)}] passIndex out of range for material \"{fogMaterial.name}\".");
            return;
        }

        _pass.ConfigureInput(ScriptableRenderPassInput.Color | ScriptableRenderPassInput.Depth);
        _pass.renderPassEvent = (RenderPassEvent)injectionPoint;
        _pass.Setup(fogMaterial, passIndex, fetchColorBuffer, bindDepthStencilAttachment);
        renderer.EnqueuePass(_pass);
    }

    protected override void Dispose(bool disposing)
    {
        _pass?.Dispose();
    }

    // ---------------- PASS ----------------
    class FullscreenFogPass : ScriptableRenderPass
    {
        // Replace internal ShaderPropertyId with our own IDs:
        private static readonly int ID_BlitTexture   = Shader.PropertyToID("_BlitTexture");
        private static readonly int ID_BlitScaleBias = Shader.PropertyToID("_BlitScaleBias");

        private Material _mat;
        private int _passIndex;
        private bool _copyActiveColor;
        private bool _bindDepthStencil;

        private RTHandle _copiedColor;
        private static readonly MaterialPropertyBlock s_MPBlock = new MaterialPropertyBlock();

        public FullscreenFogPass(string profilerName)
        {
            profilingSampler = new ProfilingSampler(profilerName);
        }

        public void Setup(Material m, int passIndex, bool copyColor, bool bindDepth)
        {
            _mat = m;
            _passIndex = passIndex;
            _copyActiveColor = copyColor;
            _bindDepthStencil = bindDepth;
        }

        public override void OnCameraSetup(CommandBuffer cmd, ref RenderingData renderingData)
        {
            ResetTarget();

            if (_copyActiveColor)
            {
                var desc = renderingData.cameraData.cameraTargetDescriptor;
                desc.msaaSamples = 1;
                desc.depthBufferBits = 0;
                RenderingUtils.ReAllocateIfNeeded(ref _copiedColor, desc, name: "_VolumetricFog_ColorCopy");
            }
        }

        public void Dispose()
        {
            _copiedColor?.Release();
        }

        public override void Execute(ScriptableRenderContext context, ref RenderingData renderingData)
        {
            // DO NOT use renderingData.commandBuffer (internal). Use our own:
            CommandBuffer cmd = CommandBufferPool.Get(profilingSampler.name);

            using (new ProfilingScope(cmd, profilingSampler))
            {
                ref var camData = ref renderingData.cameraData;

                if (_copyActiveColor)
                {
                    CoreUtils.SetRenderTarget(cmd, _copiedColor);
                    // Cheap copy of camera color → _copiedColor
                    Blitter.BlitTexture(cmd, camData.renderer.cameraColorTargetHandle, new Vector4(1, 1, 0, 0), 0.0f, false);
                }

                if (_bindDepthStencil)
                    CoreUtils.SetRenderTarget(cmd, camData.renderer.cameraColorTargetHandle, camData.renderer.cameraDepthTargetHandle);
                else
                    CoreUtils.SetRenderTarget(cmd, camData.renderer.cameraColorTargetHandle);

                s_MPBlock.Clear();
                if (_copyActiveColor)
                    s_MPBlock.SetTexture(ID_BlitTexture, _copiedColor);

                s_MPBlock.SetVector(ID_BlitScaleBias, new Vector4(1, 1, 0, 0));

                // Fullscreen triangle
                cmd.DrawProcedural(Matrix4x4.identity, _mat, _passIndex, MeshTopology.Triangles, 3, 1, s_MPBlock);
            }

            // Submit + release our command buffer
            context.ExecuteCommandBuffer(cmd);
            CommandBufferPool.Release(cmd);
        }
    }
}
